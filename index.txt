<script>
    let players = [], courses = [], rounds = [], pairsT = null, singlesT = null, club = { name: "Golf Pro Manager", logo: "" };

    function loadData() {
        players = JSON.parse(localStorage.getItem('g_players')) || [];
        courses = JSON.parse(localStorage.getItem('g_courses')) || [];
        rounds = JSON.parse(localStorage.getItem('g_rounds')) || [];
        pairsT = JSON.parse(localStorage.getItem('g_pairsT'));
        singlesT = JSON.parse(localStorage.getItem('g_singlesT'));
        club = JSON.parse(localStorage.getItem('g_club')) || { name: "Golf Pro Manager", logo: "" };
    }

    function saveData() {
        localStorage.setItem('g_players', JSON.stringify(players));
        localStorage.setItem('g_courses', JSON.stringify(courses));
        localStorage.setItem('g_rounds', JSON.stringify(rounds));
        localStorage.setItem('g_pairsT', JSON.stringify(pairsT));
        localStorage.setItem('g_singlesT', JSON.stringify(singlesT));
        localStorage.setItem('g_club', JSON.stringify(club));
    }

    function applyClubProfile() {
        document.getElementById('header-club-name').innerText = club.name;
        const img = document.getElementById('club-logo-img');
        if(club.logo) { img.src = club.logo; img.style.display = 'block'; }
        else { img.style.display = 'none'; }
    }

    function handleLogoUpload(input) {
        const file = input.files[0];
        if (!file) return;
        const reader = new FileReader();
        reader.onload = (e) => { club.logo = e.target.result; };
        reader.readAsDataURL(file);
    }

    function saveClubProfile() {
        club.name = document.getElementById('setupClubName').value;
        saveData();
        applyClubProfile();
        alert("Branding Updated!");
    }

    function showTab(id) {
        document.querySelectorAll('.tab-content').forEach(t => t.classList.add('hidden'));
        document.querySelectorAll('.nav button').forEach(b => b.classList.remove('active'));
        document.getElementById(id).classList.remove('hidden');
        document.getElementById('btn-' + id.split('-')[1]).classList.add('active');
        loadData();
        applyClubProfile();
        if(id === 'tab-dash') populateDropdown('historyDropdown');
        if(id === 'tab-stats') { populateDropdown('statsDateFilter', true); renderStats(); }
        if(id === 'tab-leaderboard') renderLeaderboard();
        if(id === 'tab-setup') { renderAdmin(); document.getElementById('setupClubName').value = club.name; }
        if(id === 'tab-pairs' || id === 'tab-singles') renderMatchplay(id.split('-')[1]);
        if(id === 'tab-comp') {
            document.getElementById('rCourse').innerHTML = courses.map(c => `<option value="${c.id}">${c.name}</option>`).join('');
            document.getElementById('rPlayer').innerHTML = players.map(p => `<option value="${p.id}">${p.name}</option>`).join('');
        }
    }

    // --- PLAYER CRUD ---
    function savePlayer() {
        const name = document.getElementById('pName').value;
        const hc = document.getElementById('pHC').value;
        const eid = document.getElementById('editPlayerId').value;
        if(!name || hc === "") return;
        if(eid) { players[players.findIndex(p=>p.id==eid)] = {id:eid, name, handicap:hc}; }
        else { players.push({id:Date.now(), name, handicap:hc}); }
        saveData(); cancelPlayerEdit(); renderAdmin();
    }
    function editPlayer(id) {
        const p = players.find(x=>x.id==id);
        document.getElementById('editPlayerId').value = p.id;
        document.getElementById('pName').value = p.name;
        document.getElementById('pHC').value = p.handicap;
        document.getElementById('playerSaveBtn').innerText = "Update Player";
        document.getElementById('cancelPlayerEditBtn').style.display = "block";
    }
    function cancelPlayerEdit() {
        document.getElementById('editPlayerId').value = "";
        document.getElementById('pName').value = "";
        document.getElementById('pHC').value = "";
        document.getElementById('playerSaveBtn').innerText = "Add Player";
        document.getElementById('cancelPlayerEditBtn').style.display = "none";
    }
    function deletePlayer(id) { if(confirm("Delete Player?")) { players = players.filter(p=>p.id!=id); saveData(); renderAdmin(); }}

    // --- MATCHPLAY ---
    function renderSelector(type) {
        document.getElementById(type + '-selector').innerHTML = players.map(p => `
            <div class="item-row">
                <span>${p.name} (HC: ${p.handicap})</span>
                <input type="checkbox" class="${type}-ck" value="${p.id}" style="width:20px">
            </div>
        `).join('');
    }
    function initTourney(type) {
        const ids = Array.from(document.querySelectorAll('.' + type + '-ck:checked')).map(i => i.value);
        if (ids.length < 2) return alert("Select players");
        let units = [];
        if (type === 'pairs') {
            const shuf = ids.sort(() => 0.5 - Math.random());
            for (let i = 0; i < shuf.length; i += 2) {
                const p1 = players.find(p => p.id == shuf[i]);
                const p2 = players.find(p => p.id == shuf[i + 1]);
                units.push({ id: 'u'+i, name: p2 ? p1.name + " / " + p2.name : p1.name + " (Solo)" });
            }
        } else {
            units = ids.map(id => ({ id: 'u'+id, name: players.find(p => p.id == id).name }));
        }
        const ms = [];
        for (let i = 0; i < units.length; i += 2) { ms.push({ id: 'm'+i+Date.now(), uA: units[i], uB: units[i+1]||null, win: units[i+1]?null:'A' }); }
        const obj = { rounds: [ms] };
        if (type === 'pairs') pairsT = obj; else singlesT = obj;
        saveData(); renderMatchplay(type);
    }
    function renderMatchplay(type) {
        const t = (type === 'pairs') ? pairsT : singlesT;
        const setup = document.getElementById(type + '-setup'), bracket = document.getElementById(type + '-bracket');
        if (!t) { setup.style.display = 'block'; bracket.innerHTML = ''; renderSelector(type); }
        else {
            setup.style.display = 'none';
            let html = `<button class="btn-del" onclick="resetT('${type}')">Reset Bracket</button>`;
            t.rounds.forEach((rd, i) => {
                html += `<h4>Round ${i+1}</h4>`;
                rd.forEach(m => {
                    html += `<div class="match-card">
                        <div class="${m.win==='A'?'winner-row':''}">${m.uA.name}</div>
                        <div style="text-align:center;font-size:10px">VS</div>
                        <div class="${m.win==='B'?'winner-row':''}">${m.uB?m.uB.name:'BYE'}</div>
                        <div style="display:flex;gap:5px;margin-top:5px">
                            <button class="btn-outline" onclick="setWin('${type}',${i},'${m.id}','A')">A Win</button>
                            ${m.uB?`<button class="btn-outline" onclick="setWin('${type}',${i},'${m.id}','B')">B Win</button>`:''}
                        </div></div>`;
                });
                if (i === t.rounds.length-1 && rd.every(m=>m.win!==null) && rd.length > 1) html += `<button class="btn" onclick="advanceRound('${type}')">Next Round</button>`;
            });
            bracket.innerHTML = html;
        }
    }
    function setWin(t,ri,mi,w) { const o=(t==='pairs')?pairsT:singlesT; o.rounds[ri].find(x=>x.id===mi).win=w; saveData(); renderMatchplay(t); }
    function advanceRound(t) {
        const o=(t==='pairs')?pairsT:singlesT; const last=o.rounds[o.rounds.length-1];
        const winners=last.map(m=>m.win==='A'?m.uA:m.uB).filter(w=>w!==null);
        const next=[]; for(let i=0;i<winners.length;i+=2) next.push({id:'m'+i+Date.now(),uA:winners[i],uB:winners[i+1]||null,win:winners[i+1]?null:'A'});
        o.rounds.push(next); saveData(); renderMatchplay(t);
    }
    function resetT(t) { if(confirm("Reset?")) { if(t==='pairs') pairsT=null; else singlesT=null; saveData(); renderMatchplay(t); }}

    // --- SCORING & ADMIN ---
    function saveRound() {
        const cId = document.getElementById('rCourse').value, pId = document.getElementById('rPlayer').value;
        const crs = courses.find(c=>c.id==cId), plr = players.find(p=>p.id==pId);
        const scs = Array.from(document.querySelectorAll('.round-scores-val')).map(i=>parseInt(i.value)||0);
        let pts = 0;
        scs.forEach((s, i) => {
            if(s>0){
                const par = parseInt(crs.pars[i]), si = parseInt(crs.si[i]), hc = parseInt(plr.handicap);
                const allow = Math.floor(hc/18) + (hc%18 >= si ? 1 : 0);
                pts += Math.max(0, par - (s - allow) + 2);
            }
        });
        rounds.push({id:Date.now(), date:document.getElementById('rDate').value, playerName:plr.name, points:pts, scores:scs});
        saveData(); showTab('tab-leaderboard');
    }
    function renderAdmin() {
        document.getElementById('admin-players').innerHTML = players.map(p => `<div class="item-row"><span>${p.name} (HC: ${p.handicap})</span><div><button class="btn-edit" onclick="editPlayer('${p.id}')">Edit</button><button class="btn-del" onclick="deletePlayer('${p.id}')">Delete</button></div></div>`).join('');
        document.getElementById('admin-courses').innerHTML = courses.map(c => `<div class="item-row">${c.name}</div>`).join('');
    }
    function saveCourse() {
        const p = Array.from(document.querySelectorAll('.setup-pars-val')).map(i=>i.value);
        const s = Array.from(document.querySelectorAll('.setup-si-val')).map(i=>i.value);
        courses.push({id:Date.now(), name:document.getElementById('cName').value, pars:p, si:s});
        saveData(); renderAdmin();
    }
    function populateDropdown(id, all=false) {
        const dates = [...new Set(rounds.map(r=>r.date))].sort().reverse();
        document.getElementById(id).innerHTML = (all ? '<option value="all">All Dates</option>' : '') + dates.map(d=>`<option value="${d}">${d}</option>`).join('');
    }
    function renderStats() {
        const f = document.getElementById('statsDateFilter').value;
        const filtered = f === 'all' ? rounds : rounds.filter(r=>r.date===f);
        document.getElementById('stats-body').innerHTML = filtered.map(r => `<tr><td>${r.playerName}</td><td>${r.date}</td><td>${r.points}</td><td><button class="btn-del" onclick="deleteRound('${r.id}')">X</button></td></tr>`).join('');
    }
    function deleteRound(id) { rounds = rounds.filter(r=>r.id!=id); saveData(); renderStats(); }
    function renderLeaderboard() {
        const t = new Date().toISOString().split('T')[0];
        const f = rounds.filter(r=>r.date===t).sort((a,b)=>b.points-a.points);
        document.getElementById('leaderboard-display').innerHTML = f.map(r=>`<div class="item-row"><span>${r.playerName}</span><b>${r.points} pts</b></div>`).join('');
    }
    function setupInputGrids() { ['round-scores', 'setup-pars', 'setup-si'].forEach(g => { const el = document.getElementById(g); for(let i=1; i<=18; i++) el.innerHTML += `<div class="hole-input"><label>${i}</label><input type="number" class="${g}-val"></div>`; }); }
    function factoryReset() { if(confirm("Wipe everything?")) { localStorage.clear(); location.reload(); }}
    window.onload = () => { loadData(); setupInputGrids(); showTab('tab-dash'); };
</script>
